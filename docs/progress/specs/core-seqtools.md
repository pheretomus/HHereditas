MODULE KEY: seqtools
OWNERSHIP: packages/core/src/seq/translate.ts, seq/codon-tables.ts, seq/orf.ts, seq/search.ts, seq/stats.ts, their *.test.ts files, tools/oracle/seqtools_*.py, packages/core/test/oracle/seqtools/**. When (and only when) your files are complete and green, APPEND export lines for them to packages/core/src/seq/index.ts (do not change existing lines or other seq files).

API:
  GENETIC_CODES: every NCBI translation table available in Biopython's Bio.Data.CodonTable, generated by tools/oracle/seqtools_codons.py into codon-tables.ts (forward table, start codons, stop codons, names); getGeneticCode(id = 1).
  translate(dna: string, options?: { table?: number; frame?: 0 | 1 | 2; toStop?: boolean; cds?: boolean }): string
     // DNA (or RNA) → one-letter protein, '*' for stops; a trailing partial codon is ignored; ambiguous IUPAC codons → the amino acid when every expansion gives the same one (e.g. GCN → A), else 'X' (match Biopython's behaviour, including B/Z/J if Biopython produces them; document). cds: the first codon becomes M when it is a start codon of the table.
  translateFeature(rec: SeqRecord, feature: Feature, options?): string   // honours strand, joined segments, /codon_start (1–3), /transl_table, wrapping segments; first codon → M for CDS features when it is a start codon of the table.
  findOrfs(seq: string, topology: Topology, options?: { minAaLength?: number /* default 100 */; table?: number; startCodons?: 'ATG' | 'table'; strands?: 'both' | 'top' | 'bottom'; includeNested?: boolean /* default false: only the longest ORF per stop codon */; allowNoStop?: boolean /* linear: ORFs running off the end, default false */ }): Orf[]   // Orf from types.ts; on circular molecules ORFs can cross the origin (range.end > L; cap at one full turn); sorted by range.start.
  searchSequence(seq: string, topology: Topology, query: string, options?: { strands?: 'both' | 'top' | 'bottom'; maxMismatches?: number /* 0..3 */; iupac?: boolean /* default true */ }): SearchHit[]   // hits across the origin on circular molecules; palindromic query hits reported once; sorted.
  searchProtein(seq, topology, peptide: string, options?: { table?: number }): Array<SearchHit & { frame: 0 | 1 | 2 }>   // 6-frame translated search, ranges in DNA coordinates.
  gcContent(seq): number   // 0..1 over unambiguous bases (S counts as GC, W as AT; document N handling)
  baseCounts(seq): Record<string, number>
  dnaMolecularWeight(seq: string, options?: { doubleStranded?: boolean; circular?: boolean; molecule?: 'DNA' | 'RNA'; monoisotopic?: boolean }): number   // Da; must equal Bio.SeqUtils.molecular_weight within 0.01 Da for the same options (check its 5'-phosphate semantics).
  proteinMolecularWeight(protein: string): number   // Da, average masses, equal to Biopython ProteinAnalysis.molecular_weight()
  proteinIsoelectricPoint(protein: string): number   // equal to Biopython IsoelectricPoint within 0.01
  proteinExtinctionCoefficient(protein: string): { reduced: number; oxidized: number }   // equal to Biopython ProteinAnalysis.molar_extinction_coefficient

ORACLE & TESTS: tools/oracle/seqtools_oracle.py → translations for every table on random sequences including ambiguous codons and all 3 frames; MW/pI/extinction for random inputs; reference ORF lists from a straightforward brute-force Python implementation you write from the definition (document it). Performance: findOrfs on the 4.6 Mb genome < 2 s; exact searchSequence on the genome < 300 ms; 2-mismatch search for a 20-mer < 3 s (skipped when the fixture is absent).
